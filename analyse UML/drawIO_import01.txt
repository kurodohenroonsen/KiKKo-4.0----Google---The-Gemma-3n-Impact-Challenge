' --- Package: Noyau de Données (Core Data Models) ---
package "1. Core Data Models" {
  class Card << (E,orchid) Entity >> {
    +id: String
    +deckName: String
    +name: String
    +stats: Map<String, Double>
    +aiOverviewHtml: String
    +bourdonTTS: String
    +provenanceLog: String
    +imagesBase64: List<String>
  }

  class QuizQuestion << (D,lightgreen) Data >> {
    +q: String
    +o: List<String>
    +c: Int
  }

  class Provenance << (D,lightblue) Embedded >> {
    +type: ProvenanceType
    +sourceName: String
    +isReproducible: Boolean
  }

  enum ProvenanceType {
    HIVE_FORGED
    HORNET_SOURCED
  }
}

package "2. Database (Room)" {
  class AppDatabase << (D,khaki) Database >> {
    +cardDao(): CardDao
  }

  interface CardDao << DAO >> {
    +insertAll(cards: List<Card>)
    +getById(id: String): Card?
    +getAll(): List<Card>
    +getCount(): Int
  }

  class TypeConverters << Utility >> {
    +fromQuizList(quiz: List<QuizQuestion>): String
    +toQuizList(json: String): List<QuizQuestion>
    +fromStatsMap(stats: Map<String, Double>): String
    +toStatsMap(json: String): Map<String, Double>
    +fromStringList(list: List<String>): String
    +toStringList(json: String): List<String>
  }
}
