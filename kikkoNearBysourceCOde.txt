    private fun observeModelDownload(model: Model) {
        WorkManager.getInstance(this).getWorkInfosForUniqueWorkLiveData(model.id)
            .observe(this) { workInfos ->
                val workInfo = workInfos?.firstOrNull() ?: return@observe
                when (workInfo.state) {
                    WorkInfo.State.RUNNING -> {
                        val progress = workInfo.progress
                        val isUnzipping = progress.getBoolean(DownloadWorker.KEY_START_UNZIPPING, false)
                        if (isUnzipping) {
                            updateModelProgress(model.id, DownloadStatus.DOWNLOADING, model.sizeInBytes, 100, isUnzipping = true)
                        } else {
                            val downloadedBytes = progress.getLong(DownloadWorker.KEY_PROGRESS_BYTES, 0L)
                            val percent = progress.getInt(DownloadWorker.KEY_PROGRESS_PERCENT, 0)
                            updateModelProgress(model.id, DownloadStatus.DOWNLOADING, downloadedBytes, percent, isUnzipping = false)
                        }
                    }
                    WorkInfo.State.SUCCEEDED -> {
                        updateModelProgress(model.id, DownloadStatus.SUCCEEDED, model.sizeInBytes, 100)
                        updateModelSelectors()
                        if (model.name.contains("Vosk", true)) initVosk()
                    }
                    WorkInfo.State.FAILED -> updateModelProgress(model.id, DownloadStatus.FAILED, 0L, 0)
                    WorkInfo.State.CANCELLED -> updateModelProgress(model.id, DownloadStatus.NOT_DOWNLOADED, 0L, 0)
                    else -> {}
                }
            }
    }

        SttService.result.observe(this) { result ->

                    buttonTestVosk.text = "Test Vosk (Écoute)"
                    if (result.isFinal) {
                        textVoskStatus.text = "Résultat final: ${result.text}"
                    }
                }
                SttStatus.ERROR -> {
                    isVoskListening = false
                    buttonTestVosk.text = "Test Vosk (Écoute)"
                    textVoskStatus.text = result.text
                }
                SttStatus.PROCESSING -> {
                    textVoskStatus.text = result.text
                }
            }
        }
    }

            }
        }
    }


    private fun handleModelAction(model: Model) {
        when (model.status) {
            DownloadStatus.NOT_DOWNLOADED, DownloadStatus.FAILED -> ModelManager.downloadModel(this, model)
            DownloadStatus.SUCCEEDED -> ModelManager.deleteModel(this, model)
            else -> Toast.makeText(this, "Action en cours...", Toast.LENGTH_SHORT).show()
        }
    }

    private fun updateModelProgress(modelId: String, status: DownloadStatus, downloadedBytes: Long, progress: Int, isUnzipping: Boolean = false) {
        val modelIndex = llmAndVoskModels.indexOfFirst { it.id == modelId }
        if (modelIndex != -1) {
            llmAndVoskModels[modelIndex].status = status
            llmAndVoskModels[modelIndex].downloadedBytes = downloadedBytes
            llmAndVoskModels[modelIndex].progress = progress
            modelAdapter.notifyItemChanged(modelIndex, ModelAdapter.ProgressPayload(isUnzipping))
        }
    }

    @SuppressLint("NotifyDataSetChanged")
    private fun loadTranslatorModels() {
        val remoteModelManager = RemoteModelManager.getInstance()
        val deviceLanguage = Locale.getDefault().language
        val targetLanguageCode = TranslateLanguage.fromLanguageTag(deviceLanguage) ?: TranslateLanguage.ENGLISH

        remoteModelManager.getDownloadedModels(TranslateRemoteModel::class.java)
            .addOnSuccessListener { downloadedModels ->
                val downloadedLanguages = downloadedModels.map { it.language }
                translatorModels.clear()
                TranslateLanguage.getAllLanguages().forEach { langCode ->
                    if (langCode != targetLanguageCode) {
                        translatorModels.add(
                            Model(
                                id = "translator-$langCode-to-$targetLanguageCode",
                                name = "Traducteur: ${Locale(langCode).displayName}",
                                description = "Traduit de '${Locale(langCode).displayLanguage}' vers '${Locale(targetLanguageCode).displayLanguage}'",
                                url = "",
                                archiveName = "mlkit_translator_${langCode}.zip",
                                modelFileName = "mlkit_translator_${langCode}",
                                sizeInBytes = 30_000_000,
                                status = if (downloadedLanguages.contains(langCode)) DownloadStatus.SUCCEEDED else DownloadStatus.NOT_DOWNLOADED
                            )
                        )
                    }
                }
                translatorAdapter.updateModels(translatorModels.sortedBy { it.name })
            }
    }

    private fun handleTranslatorAction(model: Model) {
        val remoteModelManager = RemoteModelManager.getInstance()
        val langCode = model.id.split("-")[1]
        val translatorModel = TranslateRemoteModel.Builder(langCode).build()

        if (model.status == DownloadStatus.SUCCEEDED) {
            remoteModelManager.deleteDownloadedModel(translatorModel)
                .addOnSuccessListener {
                    Toast.makeText(this, "Modèle '${model.name}' supprimé.", Toast.LENGTH_SHORT).show()
                    loadTranslatorModels()
                }
        } else {
            Toast.makeText(this, "Téléchargement du modèle '${model.name}'...", Toast.LENGTH_LONG).show()
            remoteModelManager.download(translatorModel, com.google.mlkit.common.model.DownloadConditions.Builder().build())
                .addOnSuccessListener {
                    Toast.makeText(this, "Modèle '${model.name}' téléchargé !", Toast.LENGTH_SHORT).show()
                    loadTranslatorModels()
                }
        }
    }

        val downloadedGemmaModels = llmAndVoskModels.filter { it.name.contains("Gemma", true) && it.status == DownloadStatus.SUCCEEDED }

        populateRadioGroup(radioGroupVoskModels, downloadedVoskModels, "Vosk")
        populateRadioGroup(radioGroupGemmaModels, downloadedGemmaModels, "Gemma")
    }

    private fun populateRadioGroup(radioGroup: RadioGroup, models: List<Model>, type: String) {
        val currentSelectionTag = if (radioGroup.checkedRadioButtonId != -1) {
            findViewById<RadioButton>(radioGroup.checkedRadioButtonId)?.tag as? String
        } else null
        radioGroup.removeAllViews()

        if (models.isEmpty()) {
            val noModelText = TextView(this).apply { text = "Aucun modèle $type n'est prêt." }
            radioGroup.addView(noModelText)
            return
        }

        var selectionRestored = false
        models.forEach { model ->
            val radioButton = RadioButton(this).apply {
                text = model.name
                tag = model.id
                id = View.generateViewId()
            }
            radioGroup.addView(radioButton)
            if (model.id == currentSelectionTag) {
                radioGroup.check(radioButton.id)
                selectionRestored = true
            }
        }

        if (!selectionRestored && radioGroup.childCount > 0) {
            (radioGroup.getChildAt(0) as? RadioButton)?.let {
                radioGroup.check(it.id)
            }
        }
    }

    private fun initVosk(modelId: String? = null) {
        val selectedModelId = modelId ?: if (radioGroupVoskModels.checkedRadioButtonId != -1) {
            findViewById<RadioButton>(radioGroupVoskModels.checkedRadioButtonId).tag as? String
        } else {
            llmAndVoskModels.find { it.name.contains("Vosk", true) && it.status == DownloadStatus.SUCCEEDED }?.id
        }

        if (selectedModelId == null) {
            textVoskStatus.text = "Statut Vosk: Aucun modèle téléchargé."
            return
        }

        val voskModelEntry = llmAndVoskModels.find { it.id == selectedModelId }
        if (voskModelEntry != null) {
            textVoskStatus.text = "Statut Vosk: Chargement de ${voskModelEntry.name}..."
            val modelPath = voskModelEntry.getModelLocalPath(this).absolutePath
            SttService.selectModel(modelPath) { success, error ->
                if (success) {
                    textVoskStatus.text = "Statut Vosk: Prêt (${voskModelEntry.name})."
                } else {
                    textVoskStatus.text = "Statut Vosk: Erreur ($error)"
                }
            }
        }
    }

    private fun identifyLanguage(text: String) {
        LanguageIdentification.getClient().identifyLanguage(text)
            .addOnSuccessListener { languageCode ->
                if (languageCode == "und") {
                    textTextRecognitionResult.append("\n\nLangue: Indéterminée")
                } else {
                    val deviceLanguage = Locale.getDefault().language
                    textTextRecognitionResult.append("\n\nLangue détectée: $languageCode (Appareil: $deviceLanguage)")
                    if (languageCode != deviceLanguage) {
                        translateTextIfNeeded(text, languageCode, deviceLanguage)
                    }
                }
            }
    }

    private fun translateTextIfNeeded(text: String, sourceLanguage: String, targetLanguage: String) {
        val options = TranslatorOptions.Builder()
            .setSourceLanguage(sourceLanguage)
            .setTargetLanguage(targetLanguage)
            .build()
        val translator = Translation.getClient(options)
        lifecycle.addObserver(translator)
        textTextRecognitionResult.append("\nTraduction en cours...")

        translator.downloadModelIfNeeded()
            .addOnSuccessListener {
                Log.i(TAG, "Modèle de traduction ($sourceLanguage -> $targetLanguage) prêt.")
                translator.translate(text)
                    .addOnSuccessListener { translatedText -> textTextRecognitionResult.append("\n\nTraduction:\n$translatedText") }
                    .addOnFailureListener { e -> textTextRecognitionResult.append("\n\nÉchec de la traduction: ${e.message}") }
            }
    }

package be.aktina.android.kikkopocnearby

import com.google.gson.Gson

/**
 * Représente une seule carte de "Miel Forgé".
 * C'est l'unité de connaissance de base dans notre PoC.
 */
data class HoneyCard(
    val id: String,
    val name: String, // "Abeille", "Miel", etc.
    val type: String, // "Animal" ou "Produit Transformé"
    val description: String
)

/**
 * Représente le paquet complet que deux Foragers s'échangent.
 * Il contient le nom du Forager et sa liste de cartes.
 */
data class TrustedPackage(
    val foragerName: String,
    val cards: List<HoneyCard>
) {
    // Méthode pratique pour convertir cet objet en JSON, puis en bytes pour l'envoi.
    fun toBytes(): ByteArray {
        val gson = Gson()
        return gson.toJson(this).toByteArray(Charsets.UTF_8)
    }

    // Méthode statique pour faire l'inverse : convertir des bytes en notre objet.
    companion object {
        fun fromBytes(bytes: ByteArray): TrustedPackage {
            val gson = Gson()
            return gson.fromJson(String(bytes, Charsets.UTF_8), TrustedPackage::class.java)
        }
    }
}

    private fun setupIconClickListeners() {
        gpsStatusIcon.setOnClickListener {
            if (isDiscovering) stopDiscovery() else startDiscovery()
        }
        advertisingStatusIcon.setOnClickListener {
            if (isAdvertising) stopAdvertising() else startAdvertising()
        }
    }

    @SuppressLint("MissingPermission")
    private fun onPermissionsGranted() {
        Toast.makeText(this, getString(R.string.toast_permissions_granted), Toast.LENGTH_SHORT).show()
        gpsStatusIcon.isEnabled = true
        advertisingStatusIcon.isEnabled = true
        startLocationUpdates()
        startDiscovery()
    }

    @SuppressLint("MissingPermission")
    private fun startLocationUpdates() {
        val locationRequest = LocationRequest.Builder(Priority.PRIORITY_HIGH_ACCURACY, 5000).build()
        fusedLocationProviderClient.requestLocationUpdates(locationRequest, locationCallback, Looper.getMainLooper())
    }

    private fun stopLocationUpdates() {
        fusedLocationProviderClient.removeLocationUpdates(locationCallback)
    }

    private val locationCallback = object : LocationCallback() {
        override fun onLocationResult(locationResult: LocationResult) {
            val newLocation = locationResult.lastLocation ?: return

            if (myLocation == null) {
                gpsStatusIcon.setColorFilter(colorActive, PorterDuff.Mode.SRC_IN)
                Toast.makeText(this@MainActivity, getString(R.string.toast_gps_acquired), Toast.LENGTH_SHORT).show()
                statusTextScrolling.text = getString(R.string.status_ready_gps_ok)
                if (!isAdvertising) startAdvertising()
            }

            myLocation = newLocation
            playerAdapter.updateUserLocation(myLocation)
            radarView.updatePlayers(discoveredPlayers, myLocation)

            if (isAdvertising) {
                val currentTime = System.currentTimeMillis()
                val timeSinceLastUpdate = currentTime - lastAdvertisingTime
                val distanceMoved = lastAdvertisedLocation?.distanceTo(newLocation) ?: Float.MAX_VALUE

                if (timeSinceLastUpdate > ADVERTISING_UPDATE_INTERVAL_MS && distanceMoved > ADVERTISING_UPDATE_DISTANCE_METERS) {
                    connectionsClient.stopAdvertising()
                    advertiseCurrentLocation()
                }
            }
        }
    }

    private fun startAdvertising() {
        isAdvertising = true
        advertisingStatusIcon.setColorFilter(colorActive, PorterDuff.Mode.SRC_IN)
        statusTextScrolling.text = getString(R.string.status_visible)
        startLocationUpdates()
        advertiseCurrentLocation()
    }

    private fun stopAdvertising() {
        isAdvertising = false
        connectionsClient.stopAdvertising()
        advertisingStatusIcon.setColorFilter(colorInactive, PorterDuff.Mode.SRC_IN)
        statusTextScrolling.text = getString(R.string.status_ready)
    }

    @SuppressLint("MissingPermission")
    private fun advertiseCurrentLocation() {
        if (myLocation == null) {
            fusedLocationProviderClient.lastLocation.addOnSuccessListener { location ->
                if(location != null) { myLocation = location; advertiseCurrentLocation() }
                else { Toast.makeText(this, getString(R.string.toast_gps_unavailable), Toast.LENGTH_SHORT).show() }
            }
            return
        }

        val deckCountsCsv = myCatalogue.decks.joinToString(",") { myCatalogue.decks.find { d -> d.name == it.name }?.cardCount.toString() }
        val recordCsv = "${myCatalogue.wins},${myCatalogue.losses}"
        val latFormatted = String.format(Locale.US, "%.6f", myLocation!!.latitude)
        val lonFormatted = String.format(Locale.US, "%.6f", myLocation!!.longitude)
        val endpointNameCompact = "$myName|$deckCountsCsv|$recordCsv|$latFormatted $lonFormatted"

        val advertisingOptions = AdvertisingOptions.Builder().setStrategy(strategy).build()
        connectionsClient.startAdvertising(endpointNameCompact, serviceId, connectionLifecycleCallback, advertisingOptions)
            .addOnSuccessListener {
                lastAdvertisingTime = System.currentTimeMillis()
                lastAdvertisedLocation = myLocation
            }
            .addOnFailureListener { e ->
                Log.w("Nearby", "Échec temporaire de la relance de la publicité.", e)
            }
    }

    private fun startDiscovery() {
        isDiscovering = true
        gpsStatusIcon.setColorFilter(colorActive, PorterDuff.Mode.SRC_IN)
        statusTextScrolling.text = getString(R.string.status_searching)
        discoveredPlayers.clear()
        playerAdapter.notifyDataSetChanged()
        radarView.updatePlayers(discoveredPlayers, myLocation)
        radarView.visibility = View.VISIBLE
        playersRecyclerView.visibility = View.VISIBLE

        val discoveryOptions = DiscoveryOptions.Builder().setStrategy(strategy).build()
        connectionsClient.startDiscovery(serviceId, endpointDiscoveryCallback, discoveryOptions)
            .addOnFailureListener {
                isDiscovering = false
                gpsStatusIcon.setColorFilter(colorError, PorterDuff.Mode.SRC_IN)
            }
    }

    private fun stopDiscovery() {
        isDiscovering = false
        connectionsClient.stopDiscovery()
        gpsStatusIcon.setColorFilter(colorInactive, PorterDuff.Mode.SRC_IN)
        statusTextScrolling.text = getString(R.string.status_ready)
        discoveredPlayers.clear()
        playerAdapter.notifyDataSetChanged()
        radarView.visibility = View.GONE
        playersRecyclerView.visibility = View.GONE
    }

    private val endpointDiscoveryCallback = object : EndpointDiscoveryCallback() {
        override fun onEndpointFound(endpointId: String, discoveredEndpointInfo: DiscoveredEndpointInfo) {
            val compactName = discoveredEndpointInfo.endpointName
            try {
                val parts = compactName.split('|')
                if (parts.size == 4) {
                    val name = parts[0]
                    val deckCountsCsv = parts[1].split(',')
                    val recordParts = parts[2].split(',')
                    val locationParts = parts[3].split(' ')

                    val wins = recordParts[0].toInt()
                    val losses = recordParts[1].toInt()
                    val lat = locationParts[0].toDouble()
                    val lon = locationParts[1].toDouble()

                    val reconstructedDecks = mutableListOf<Deck>()
                    deckCountsCsv.forEachIndexed { index, countStr ->
                        val count = countStr.toInt()
                        if (count > 0 && index < GameConstants.MASTER_DECK_LIST.size) {
                            reconstructedDecks.add(Deck(GameConstants.MASTER_DECK_LIST[index], count))
                        }
                    }

                    val colorIndex = abs(endpointId.hashCode()) % turtleColorPalette.size
                    val assignedColor = turtleColorPalette[colorIndex]

                    val partialCatalogue = PlayerCatalogue(name, reconstructedDecks, lat, lon, wins, losses, assignedColor)

                    if (!discoveredPlayers.containsKey(endpointId)) {
                        discoveredPlayers[endpointId] = partialCatalogue
                        playerAdapter.notifyDataSetChanged()
                        radarView.updatePlayers(discoveredPlayers, myLocation)
                    }
                }
            } catch (e: Exception) {
                Log.e("Nearby", "Erreur en analysant le nom compact: $compactName", e)
            }
        }

        override fun onEndpointLost(endpointId: String) {
            discoveredPlayers.remove(endpointId)
            playerAdapter.notifyDataSetChanged()
            radarView.updatePlayers(discoveredPlayers, myLocation)
        }
    }

    private val connectionLifecycleCallback = object : ConnectionLifecycleCallback() {
        override fun onConnectionInitiated(endpointId: String, connectionInfo: ConnectionInfo) {
            val opponentNameFromInfo = connectionInfo.endpointName.split('|').firstOrNull() ?: "Adversaire"

            // BOURDON: On applique notre nouveau thème de dialogue ici !
            AlertDialog.Builder(this@MainActivity, R.style.KikkoAlertDialogTheme)
                .setTitle(getString(R.string.dialog_connection_title, opponentNameFromInfo))
                .setMessage(getString(R.string.dialog_connection_message, connectionInfo.authenticationDigits))
                .setPositiveButton(getString(R.string.dialog_accept)) { _, _ ->
                    connectionsClient.acceptConnection(endpointId, payloadCallback)
                }
                .setNegativeButton(getString(R.string.dialog_decline)) { _, _ -> connectionsClient.rejectConnection(endpointId) }
                .show()
        }

        override fun onConnectionResult(endpointId: String, result: ConnectionResolution) {
            val name = opponentCatalogue?.playerName ?: "Adversaire"
            when (result.status.statusCode) {
                ConnectionsStatusCodes.STATUS_OK -> {
                    if (isAdvertising) stopAdvertising()
                    if (isDiscovering) stopDiscovery()
                    statusTextScrolling.text = getString(R.string.status_connected_to, name)
                    connectedEndpointId = endpointId
                    sendCatalogue(endpointId)
                    radarView.visibility = View.GONE
                    playersRecyclerView.visibility = View.GONE
                }
                ConnectionsStatusCodes.STATUS_CONNECTION_REJECTED -> Log.w("Nearby", "Connexion rejetée par $name")
                else -> Log.w("Nearby", "Erreur de connexion avec $name")
            }
        }

        override fun onDisconnected(endpointId: String) {
            val name = opponentCatalogue?.playerName ?: "l'adversaire"
            statusTextScrolling.text = getString(R.string.status_disconnected)
            connectedEndpointId = null
        }
    }

    private val payloadCallback = object : PayloadCallback() {
        override fun onPayloadReceived(endpointId: String, payload: Payload) {
            if (payload.type == Payload.Type.BYTES) {
                val catalogueJson = String(payload.asBytes()!!, StandardCharsets.UTF_8)
                try {
                    val fullCatalogue = gson.fromJson(catalogueJson, PlayerCatalogue::class.java)
                    opponentCatalogue = fullCatalogue
                    Toast.makeText(this@MainActivity, getString(R.string.toast_catalogue_received, fullCatalogue.playerName), Toast.LENGTH_LONG).show()
                } catch (e: Exception) {
                    Log.e("Nearby", "Erreur en décodant le payload du catalogue complet", e)
                }
            }
        }
        override fun onPayloadTransferUpdate(endpointId: String, update: PayloadTransferUpdate) {}
    }

    private fun sendCatalogue(endpointId: String) {
        myCatalogue = myCatalogue.copy(latitude = myLocation?.latitude, longitude = myLocation?.longitude)
        val catalogueJson = gson.toJson(myCatalogue)
        val payload = Payload.fromBytes(catalogueJson.toByteArray(StandardCharsets.UTF_8))
        connectionsClient.sendPayload(endpointId, payload)
    }

    private fun hasAllPermissions(): Boolean = REQUIRED_PERMISSIONS.all {
        ContextCompat.checkSelfPermission(this, it) == PackageManager.PERMISSION_GRANTED
    }

    private fun checkAndRequestPermissions() {
        if (!hasAllPermissions()) {
            requestMultiplePermissions.launch(REQUIRED_PERMISSIONS)
        } else {
            onPermissionsGranted()
        }
    }

    override fun onStop() {
        super.onStop()
        connectionsClient.stopAllEndpoints()
        if (isAdvertising) stopAdvertising()
        if (isDiscovering) stopDiscovery()
        connectedEndpointId = null
    }

    private fun resolveThemeColor(@AttrRes attr: Int): Int {
        val typedValue = TypedValue()
        theme.resolveAttribute(attr, typedValue, true)
        return typedValue.data
    }
}

package be.aktina.android.kikkopocnearby

import android.graphics.PorterDuff
import android.location.Location
import android.view.LayoutInflater

import android.view.ViewGroup

class PlayerAdapter(
    private val players: Map<String, PlayerCatalogue>,
    private var currentUserLocation: Location?,
    private val onClick: (endpointId: String, catalogue: PlayerCatalogue) -> Unit
) : RecyclerView.Adapter<PlayerAdapter.PlayerViewHolder>() {

    class PlayerViewHolder(view: View) : RecyclerView.ViewHolder(view) {
        val playerIcon: ImageView = view.findViewById(R.id.player_icon)
        val playerNameText: TextView = view.findViewById(R.id.player_name_text)
        val decksColumn1Text: TextView = view.findViewById(R.id.decks_column_1_text)
        val decksColumn2Text: TextView = view.findViewById(R.id.decks_column_2_text)
        val playerDistanceText: TextView = view.findViewById(R.id.player_distance_text)
        val playerRecordText: TextView = view.findViewById(R.id.player_record_text)
    }

    fun updateUserLocation(location: Location?) {
        this.currentUserLocation = location
        notifyDataSetChanged()
    }

    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): PlayerViewHolder {
        val view = LayoutInflater.from(parent.context)
            .inflate(R.layout.item_player_card, parent, false)
        return PlayerViewHolder(view)
    }

    override fun onBindViewHolder(holder: PlayerViewHolder, position: Int) {
        val playerEntry = players.entries.toList()[position]
        val endpointId = playerEntry.key
        val catalogue = playerEntry.value
        // BOURDON: On récupère le contexte pour accéder aux ressources string
        val context = holder.itemView.context

        holder.playerNameText.text = catalogue.playerName

        val column1Decks = StringBuilder()
        val column2Decks = StringBuilder()

        catalogue.decks.forEach { deck ->
            val emoji = GameConstants.DECK_EMOJIS[deck.name] ?: "❓"
            val deckString = "$emoji(${deck.cardCount})\n"
            when (deck.name) {
                "Animaux", "Plantes" -> column1Decks.append(deckString)
                "Recette", "Aliment" -> column2Decks.append(deckString)
            }
        }

        holder.decksColumn1Text.text = column1Decks.toString().trim()
        holder.decksColumn2Text.text = column2Decks.toString().trim()

        // BOURDON: Utilisation des ressources string pour les stats et la distance
        holder.playerRecordText.text = context.getString(R.string.card_record_format, catalogue.wins, catalogue.losses)

        if (currentUserLocation != null && catalogue.latitude != null && catalogue.longitude != null) {
            val playerLocation = Location("player").apply {
                latitude = catalogue.latitude
                longitude = catalogue.longitude
            }
            val distance = currentUserLocation!!.distanceTo(playerLocation)
            holder.playerDistanceText.text = context.getString(R.string.card_distance_format, distance)
        } else {
            holder.playerDistanceText.text = ""
        }

        holder.playerIcon.setColorFilter(catalogue.color, PorterDuff.Mode.SRC_IN)

        holder.itemView.setOnClickListener { onClick(endpointId, catalogue) }
    }

    override fun getItemCount() = players.size
}

class RadarView @JvmOverloads constructor(

) : View(context, attrs, defStyleAttr), SensorEventListener {

    // BOURDON: Paramètres de configuration pour fignoler le design.
    // Vous pouvez jouer avec ces valeurs (en dp).
    private val TURTLE_ICON_SIZE_DP = 36f
    private val CARDINAL_POINTS_RADIUS_OFFSET_DP = 25f
    private val DISTANCE_LABEL_VERTICAL_OFFSET_DP = 7f
    private val PLAYER_NAME_VERTICAL_OFFSET_DP = 12f

    // -- Variables converties en pixels pour le dessin
    private val turtleIconSizePx: Int
    private val cardinalPointsRadiusOffsetPx: Float
    private val distanceLabelVerticalOffsetPx: Float
    private val playerNameVerticalOffsetPx: Float

    // -- Paints
    private val playerTextPaint: Paint = Paint(Paint.ANTI_ALIAS_FLAG)
    private val cardinalPaint: Paint = Paint(Paint.ANTI_ALIAS_FLAG)
    private val labelPaint: Paint = Paint(Paint.ANTI_ALIAS_FLAG)
    private val turtleBitmap: Bitmap

    // -- Données des capteurs et de jeu
    private val sensorManager: SensorManager = context.getSystemService(Context.SENSOR_SERVICE) as SensorManager
    private val accelerometerReading = FloatArray(3)
    private val magnetometerReading = FloatArray(3)
    private var smoothedAccelerometerReading = FloatArray(3)
    private var smoothedMagnetometerReading = FloatArray(3)
    private val rotationMatrix = FloatArray(9)
    private val orientationAngles = FloatArray(3)
    private var azimuth: Float = 0f
    private var players: List<PlayerCatalogue> = emptyList()
    private var userLocation: Location? = null
    private val maxDistanceMeters = 20f
    private val cardinalPoints = mapOf("N" to 0f, "E" to 90f, "S" to 180f, "O" to 270f)
    private val alpha: Float = 0.1f

    init {
        val density = context.resources.displayMetrics.density
        // Conversion des DP en Px
        turtleIconSizePx = (TURTLE_ICON_SIZE_DP * density).toInt()
        cardinalPointsRadiusOffsetPx = CARDINAL_POINTS_RADIUS_OFFSET_DP * density
        distanceLabelVerticalOffsetPx = DISTANCE_LABEL_VERTICAL_OFFSET_DP * density
        playerNameVerticalOffsetPx = PLAYER_NAME_VERTICAL_OFFSET_DP * density

        val primaryColor = resolveThemeColor(com.google.android.material.R.attr.colorPrimary)
        val onSurfaceColor = resolveThemeColor(com.google.android.material.R.attr.colorOnSurface)

        playerTextPaint.color = onSurfaceColor
        playerTextPaint.textSize = 28f // Un peu plus petit
        playerTextPaint.textAlign = Paint.Align.CENTER

        cardinalPaint.color = primaryColor
        cardinalPaint.textSize = 52f
        cardinalPaint.textAlign = Paint.Align.CENTER
        cardinalPaint.isFakeBoldText = true

        labelPaint.color = onSurfaceColor
        labelPaint.textSize = 30f // Un peu plus petit
        labelPaint.textAlign = Paint.Align.CENTER
        labelPaint.isFakeBoldText = true

        val turtleDrawable = AppCompatResources.getDrawable(context, R.drawable.ic_turtle)
        turtleBitmap = turtleDrawable?.toBitmap(turtleIconSizePx, turtleIconSizePx, Bitmap.Config.ARGB_8888)!!
    }

    private fun resolveThemeColor(@AttrRes attr: Int): Int {
        val typedValue = TypedValue()
        context.theme.resolveAttribute(attr, typedValue, true)
        return typedValue.data
    }

    fun updatePlayers(newPlayers: Map<String, PlayerCatalogue>, currentUserLocation: Location?) {
        this.players = newPlayers.values.toList()
        this.userLocation = currentUserLocation
        invalidate()
    }

    override fun onDraw(canvas: Canvas) {
        super.onDraw(canvas)

        val centerX = width / 2f
        val centerY = height / 2f
        val radius = min(centerX, centerY) * 0.55f // Un peu plus grand

        // On ne dessine que les étiquettes, le fond est l'image
        val distances = listOf(5f, 10f, 15f, 20f)
        distances.forEach { distance ->
            val circleRadius = (distance / maxDistanceMeters) * radius
            canvas.drawText(
                "${distance.toInt()}m",
                centerX,
                centerY - circleRadius + labelPaint.textSize + distanceLabelVerticalOffsetPx,
                labelPaint
            )
        }

        // Dessin des points cardinaux
        val textRadius = radius + cardinalPointsRadiusOffsetPx
        cardinalPoints.forEach { (point, angle) ->
            val angleOnRadar = (angle - Math.toDegrees(azimuth.toDouble())).toFloat()
            val angleInRad = Math.toRadians(angleOnRadar.toDouble())
            val x = centerX + (textRadius * sin(angleInRad)).toFloat()
            val y = centerY - (textRadius * cos(angleInRad)).toFloat() + (cardinalPaint.textSize / 2)
            canvas.drawText(point, x, y, cardinalPaint)
        }

        // Dessin des joueurs
        val currentUserLoc = userLocation ?: return
        for (player in players) {
            val lat = player.latitude
            val lon = player.longitude

            if (lat != null && lon != null) {
                val playerLocation = Location("player").apply { latitude = lat; longitude = lon }
                val distance = currentUserLoc.distanceTo(playerLocation)
                if (distance <= maxDistanceMeters) {
                    val bearing = currentUserLoc.bearingTo(playerLocation)
                    val angleOnRadar = (bearing - Math.toDegrees(azimuth.toDouble())).toFloat()
                    val angleInRad = Math.toRadians(angleOnRadar.toDouble())
                    val distanceOnRadar = (distance / maxDistanceMeters) * radius

                    val playerX = centerX + (distanceOnRadar * sin(angleInRad)).toFloat()
                    val playerY = centerY - (distanceOnRadar * cos(angleInRad)).toFloat()

                    val paint = Paint()
                    paint.colorFilter = PorterDuffColorFilter(player.color, PorterDuff.Mode.SRC_IN)
                    canvas.drawBitmap(turtleBitmap, playerX - turtleBitmap.width / 2, playerY - turtleBitmap.height / 2, paint)

                    canvas.drawText(player.playerName, playerX, playerY + turtleBitmap.height / 2 + playerNameVerticalOffsetPx, playerTextPaint)
                }
            }
        }
    }

    override fun onAttachedToWindow() {
        super.onAttachedToWindow()
        sensorManager.getDefaultSensor(Sensor.TYPE_ACCELEROMETER)?.also {
            sensorManager.registerListener(this, it, SensorManager.SENSOR_DELAY_UI)
        }
        sensorManager.getDefaultSensor(Sensor.TYPE_MAGNETIC_FIELD)?.also {
            sensorManager.registerListener(this, it, SensorManager.SENSOR_DELAY_UI)
        }
    }

    override fun onDetachedFromWindow() {
        super.onDetachedFromWindow()
        sensorManager.unregisterListener(this)
    }

    override fun onSensorChanged(event: SensorEvent) {
        if (event.sensor.type == Sensor.TYPE_ACCELEROMETER) {
            smoothedAccelerometerReading = lowPass(event.values, smoothedAccelerometerReading)
        } else if (event.sensor.type == Sensor.TYPE_MAGNETIC_FIELD) {
            smoothedMagnetometerReading = lowPass(event.values, smoothedMagnetometerReading)
        }
        updateOrientationAngles()
    }

    override fun onAccuracyChanged(sensor: Sensor?, accuracy: Int) { }

    private fun lowPass(input: FloatArray, output: FloatArray): FloatArray {
        for (i in input.indices) {
            output[i] = output[i] + alpha * (input[i] - output[i])
        }
        return output
    }

    private fun updateOrientationAngles() {
        SensorManager.getRotationMatrix(rotationMatrix, null, smoothedAccelerometerReading, smoothedMagnetometerReading)
        SensorManager.getOrientation(rotationMatrix, orientationAngles)
        azimuth = orientationAngles[0]
        invalidate()
    }
}

package be.aktina.android.kikkopocnearby.data

import android.content.Context
import android.util.Log
import androidx.lifecycle.LiveData
import androidx.lifecycle.MutableLiveData
import androidx.work.Constraints
import androidx.work.Data
import androidx.work.ExistingWorkPolicy
import androidx.work.NetworkType
import androidx.work.OneTimeWorkRequestBuilder
import androidx.work.WorkManager
import be.aktina.android.kikkopocnearby.workers.DownloadWorker

object ModelManager {
    private const val TAG = "ModelManager"

    private val _models = MutableLiveData<List<Model>>()
    val models: LiveData<List<Model>> = _models

    fun initialize(context: Context) {
        val allModels = ModelCatalogue.getAllModels()
        val currentModels = allModels.map { model ->
            val finalModelFile = model.getModelLocalPath(context)

            if (finalModelFile.exists() && (finalModelFile.isDirectory || finalModelFile.length() > 0)) {
                model.status = DownloadStatus.SUCCEEDED
                model.downloadedBytes = model.sizeInBytes
                model.progress = 100
            } else {
                model.status = DownloadStatus.NOT_DOWNLOADED
                model.downloadedBytes = 0L
                model.progress = 0
            }
            model
        }
        _models.postValue(currentModels)
    }

    fun downloadModel(context: Context, model: Model) {
        Log.d(TAG, "Lancement du téléchargement pour le modèle: ${model.name}")
        val workManager = WorkManager.getInstance(context.applicationContext)

        val constraints = Constraints.Builder()
            .setRequiredNetworkType(NetworkType.CONNECTED)
            .build()

        val dataBuilder = Data.Builder()
        dataBuilder.putString(DownloadWorker.KEY_MODEL_ID, model.id)
        dataBuilder.putString(DownloadWorker.KEY_DOWNLOAD_URL, model.url)
        dataBuilder.putString(DownloadWorker.KEY_ARCHIVE_PATH, model.getArchiveLocalPath(context).absolutePath)
        // FRELON-ARCHITECTE: Correction de la clé et de la propriété. isZip -> isArchive
        dataBuilder.putBoolean(DownloadWorker.KEY_IS_ARCHIVE, model.isArchive)

        val downloadWorkRequest = OneTimeWorkRequestBuilder<DownloadWorker>()
            .setInputData(dataBuilder.build())
            .setConstraints(constraints)
            .addTag(model.id)
            .build()

        workManager.enqueueUniqueWork(
            model.id,
            ExistingWorkPolicy.REPLACE,
            downloadWorkRequest
        )
    }

    fun deleteModel(context: Context, model: Model) {
        Log.d(TAG, "Suppression du modèle: ${model.name}")
        val modelDir = model.getArchiveLocalPath(context.applicationContext).parentFile
        if (modelDir != null && modelDir.exists() && modelDir.isDirectory) {
            if (modelDir.deleteRecursively()) {
                Log.i(TAG, "Dossier modèle ${modelDir.name} supprimé avec succès.")
            } else {
                Log.e(TAG, "Échec de la suppression du dossier modèle ${modelDir.name}.")
            }
        }
        WorkManager.getInstance(context.applicationContext).cancelUniqueWork(model.id)
        val currentList = _models.value ?: return
        val newList = currentList.map {
            if (it.id == model.id) {
                it.copy(status = DownloadStatus.NOT_DOWNLOADED, progress = 0, downloadedBytes = 0L)
            } else {
                it
            }
        }
        _models.postValue(newList)
    }
}

/**
 * FRELON-ARCHITECTE: Un repository dédié au "butinage" de la page des modèles Vosk.
 * Il se charge de la requête réseau et de l'extraction des données depuis le HTML.
 */
object VoskModelRepository {

    private const val TAG = "VoskModelRepository"
    private const val VOSK_MODELS_URL = "https://alphacephei.com/vosk/models"

    suspend fun fetchSmallModels(): List<Model> {
        return withContext(Dispatchers.IO) {
            val models = mutableListOf<Model>()
            try {
                Log.d(TAG, "Connexion à $VOSK_MODELS_URL pour extraire les modèles Vosk...")
                val doc = Jsoup.connect(VOSK_MODELS_URL).get()
                val tables = doc.select("table")

                tables.forEach { table ->
                    val rows = table.select("tr")
                    rows.forEach { row ->
                        val columns = row.select("td")
                        if (columns.size >= 2) {
                            val linkElement = columns[0].select("a").first()
                            val modelName = linkElement?.text()
                            val modelUrl = linkElement?.attr("abs:href")
                            val sizeStr = columns[1].text()
                            val notes = columns.getOrNull(3)?.text() ?: "Modèle de reconnaissance vocale Vosk."

                            if (modelName != null && modelUrl != null && modelName.contains("small", ignoreCase = true)) {
                                val sizeInBytes = parseSize(sizeStr)
                                val fileName = modelUrl.substring(modelUrl.lastIndexOf('/') + 1)
                                val unzipDir = fileName.removeSuffix(".zip")

                                if (models.none { it.id == modelName }) {
                                    models.add(
                                        Model(
                                            id = modelName,
                                            name = "Vosk - $modelName",
                                            description = notes,
                                            url = modelUrl,
                                            archiveName = fileName,
                                            modelFileName = unzipDir,
                                            sizeInBytes = sizeInBytes,
                                            // FRELON-ARCHITECTE: Correction de la clé
                                            isArchive = true
                                        )
                                    )
                                }
                            }
                        }
                    }
                }
                Log.i(TAG, "${models.size} modèles Vosk 'small' uniques ont été identifiés pour recrutement.")
            } catch (e: Exception) {
                Log.e(TAG, "Erreur lors de la récupération des modèles Vosk", e)
            }
            models
        }
    }

    private fun parseSize(sizeStr: String): Long {
        return try {
            val value = sizeStr.replace(Regex("[^\\d.]"), "").toDoubleOrNull() ?: 0.0
            when {
                sizeStr.uppercase(Locale.ROOT).endsWith("M") -> (value * 1024 * 1024).toLong()
                sizeStr.uppercase(Locale.ROOT).endsWith("G") -> (value * 1024 * 1024 * 1024).toLong()
                else -> 0L
            }
        } catch (e: NumberFormatException) {
            Log.w(TAG, "Impossible de parser la taille: '$sizeStr'")
            0L
        }
    }
}

enum class LlmStatus {

    THINKING,
    ERROR
}

data class LlmResult(
    val status: LlmStatus,
    val text: String = ""
)

@SuppressLint("StaticFieldLeak")
object LlmService {

    private const val TAG = "LlmService"
    private var llmInference: LlmInference? = null
    private var session: LlmInferenceSession? = null
    private var currentModelPath: String? = null
    private var applicationContext: Context? = null

    private val _result = MutableLiveData<LlmResult>()
    val result: LiveData<LlmResult> = _result

    fun initialize(context: Context) {
        applicationContext = context.applicationContext
        Log.v(TAG, "Service LLM (Reine IA) prêt à recevoir un modèle.")
        _result.postValue(LlmResult(LlmStatus.IDLE, "Reine IA en attente."))
    }

    fun selectModel(context: Context, model: Model, onFinished: (Boolean, String?) -> Unit) {
        val ctx = context.applicationContext
        _result.postValue(LlmResult(LlmStatus.THINKING, "La Reine prépare ses neurones..."))
        CoroutineScope(Dispatchers.IO).launch {
            try {
                val modelPath = model.getModelLocalPath(ctx).absolutePath
                if (currentModelPath == modelPath && llmInference != null && session != null) {
                    withContext(Dispatchers.Main) {
                        _result.postValue(LlmResult(LlmStatus.IDLE, "Reine IA prête. (${model.name})"))
                        onFinished(true, null)
                    }
                    return@launch
                }

                shutdown()

                val options = LlmInference.LlmInferenceOptions.builder()
                    .setModelPath(modelPath)
                    .build()
                llmInference = LlmInference.createFromOptions(ctx, options)

                // CORRECTION 1 : On crée les options de session, y compris pour la vision
                val sessionOptions = LlmInferenceSession.LlmInferenceSessionOptions.builder()
                    .setGraphOptions(
                        GraphOptions.builder()
                            .setEnableVisionModality(model.isMultimodal)
                            .build()
                    ).build()

                // CORRECTION 1 (suite) : On passe bien les DEUX arguments
                session = LlmInferenceSession.createFromOptions(llmInference!!, sessionOptions)
                currentModelPath = modelPath

                withContext(Dispatchers.Main) {
                    _result.postValue(LlmResult(LlmStatus.IDLE, "Reine IA prête. (${model.name})"))
                    onFinished(true, null)
                }
            } catch (e: Exception) {
                val errorMessage = "Échec du chargement du modèle Gemma: ${e.message}"
                Log.e(TAG, errorMessage, e)
                shutdown()
                withContext(Dispatchers.Main) {
                    _result.postValue(LlmResult(LlmStatus.ERROR, errorMessage))
                    onFinished(false, errorMessage)
                }
            }
        }
    }

    fun generateResponse(prompt: String) {
        if (session == null) {
            _result.postValue(LlmResult(LlmStatus.ERROR, "La Reine IA n'a pas de session active."))
            return
        }

        _result.postValue(LlmResult(LlmStatus.THINKING, "réfléchit..."))

        CoroutineScope(Dispatchers.IO).launch {
            try {
                // CORRECTION 2 : La génération se fait en deux temps.
                // 1. On ajoute le prompt à la session.
                session!!.addQueryChunk(prompt)
                // 2. On appelle generateResponse SANS argument.
                val response = session!!.generateResponse()

                _result.postValue(LlmResult(LlmStatus.IDLE, response))
            } catch (e: Exception) {
                val errorMessage = "Erreur d'inférence de la Reine: ${e.message}"
                Log.e(TAG, errorMessage, e)
                _result.postValue(LlmResult(LlmStatus.ERROR, errorMessage))
            }
        }
    }

    fun shutdown() {
        Log.v(TAG, "Arrêt du service LLM (Reine IA).")
        session?.close()
        llmInference?.close()
        session = null
        llmInference = null
        currentModelPath = null
    }
}

package be.aktina.android.kikkopocnearby.services

import android.content.Context
import android.util.Log
import androidx.lifecycle.LiveData
import androidx.lifecycle.MutableLiveData
import kotlinx.coroutines.CoroutineScope
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.launch
import kotlinx.coroutines.withContext
import org.json.JSONObject
import org.vosk.Model
import org.vosk.Recognizer
import org.vosk.android.RecognitionListener
import org.vosk.android.SpeechService
import java.io.IOException

/**
 * FRELON-ARCHITECTE: Un enum pour les statuts du service STT (Speech-to-Text).
 */
enum class SttStatus {
    IDLE,       // Inactif
    LISTENING,  // Écoute en cours
    PROCESSING, // Traitement d'un résultat partiel/final
    ERROR       // Une erreur est survenue
}

/**
 * FRELON-ARCHITECTE: Une classe de données pour encapsuler le résultat du STT.
 */
data class SttResult(
    val status: SttStatus,
    val text: String = "",
    val isFinal: Boolean = false
)

/**
 * FRELON-ARCHITECTE: Service singleton pour gérer la reconnaissance vocale (STT).
 * C'est notre "Abeille-Écoutante". Elle encapsule l'initialisation et l'utilisation
 * de Vosk, le rendant réutilisable et facile à gérer.
 */
object SttService : RecognitionListener {

    private const val TAG = "SttService"
    private var model: Model? = null
    private var speechService: SpeechService? = null
    private var currentModelPath: String? = null // FRELON-ARCHITECTE: Ajout pour suivre le chemin du modèle chargé

    private val _result = MutableLiveData<SttResult>()
    val result: LiveData<SttResult> = _result

    private var currentFinalText: String = ""

    fun initialize(context: Context) {
        // L'initialisation se fait via `selectModel` maintenant
        Log.v(TAG, "Service STT prêt à recevoir un modèle.")
    }

    /**
     * Sélectionne et charge un modèle Vosk. Doit être appelé avant `startListening`.
     * Cette opération peut être longue et est donc exécutée dans une coroutine.
     */
    fun selectModel(modelPath: String, onFinished: (Boolean, String?) -> Unit) {
        CoroutineScope(Dispatchers.IO).launch {
            try {
                // FRELON-ARCHITECTE: Correction de la comparaison pour le modèle déjà chargé
                if (currentModelPath == modelPath && model != null) {
                    Log.d(TAG, "Modèle Vosk déjà chargé: $modelPath")
                    withContext(Dispatchers.Main) { onFinished(true, null) }
                    return@launch
                }
                Log.d(TAG, "Chargement du modèle Vosk depuis: $modelPath")
                model?.close() // S'assurer de fermer l'ancien modèle si un nouveau est chargé
                model = Model(modelPath)
                currentModelPath = modelPath // FRELON-ARCHITECTE: Stocke le chemin du modèle
                withContext(Dispatchers.Main) { onFinished(true, null) }
            } catch (e: IOException) {
                val errorMessage = "Échec du chargement du modèle Vosk: ${e.message}"
                Log.e(TAG, errorMessage, e)
                withContext(Dispatchers.Main) { onFinished(false, errorMessage) }
            }
        }
    }

    /**
     * Démarre l'écoute du microphone.
     */
    fun startListening() {
        if (model == null) {
            _result.postValue(SttResult(SttStatus.ERROR, "Modèle non initialisé"))
            return
        }
        if (speechService != null) {
            Log.w(TAG, "L'écoute est déjà en cours.")
            return
        }

        try {
            currentFinalText = ""
            val recognizer = Recognizer(model, 16000.0f)
            speechService = SpeechService(recognizer, 16000.0f)
            speechService?.startListening(this)
            _result.postValue(SttResult(SttStatus.LISTENING, "Écoute..."))
        } catch (e: IOException) {
            onError(e)
        }
    }

    /**
     * Arrête l'écoute du microphone.
     */
    fun stopListening() {
        speechService?.stop()
        speechService = null // Important pour indiquer que l'écoute est terminée
        _result.postValue(SttResult(SttStatus.IDLE, currentFinalText, isFinal = true))
    }

    /**
     * Libère les ressources du service STT.
     */
    fun shutdown() {
        Log.v(TAG, "Arrêt du service STT.")
        speechService?.stop()
        speechService?.shutdown()
        model?.close() // S'assurer de bien fermer le modèle Vosk
        model = null
        currentModelPath = null // Réinitialiser le chemin du modèle
    }

    // --- Implémentation de RecognitionListener ---

    override fun onPartialResult(hypothesis: String?) {
        hypothesis?.let {
            try {
                val partialText = JSONObject(it).getString("partial")
                _result.postValue(SttResult(SttStatus.PROCESSING, "$currentFinalText $partialText"))
            } catch (e: Exception) {
                Log.e(TAG, "Erreur parsing JSON partiel Vosk", e)
            }
        }
    }

    override fun onResult(hypothesis: String?) {
        hypothesis?.let {
            try {
                val segmentText = JSONObject(it).getString("text")
                if (segmentText.isNotBlank()) {
                    currentFinalText += "$segmentText "
                    _result.postValue(SttResult(SttStatus.PROCESSING, currentFinalText))
                }
            } catch (e: Exception) {
                Log.e(TAG, "Erreur parsing JSON final Vosk", e)
            }
        }
    }

    override fun onFinalResult(hypothesis: String?) {
        onResult(hypothesis)
        speechService = null
        _result.postValue(SttResult(SttStatus.IDLE, currentFinalText.trim(), isFinal = true))
        Log.i(TAG, "Vosk onFinalResult. Texte final: \"$currentFinalText\"")
    }

    override fun onError(exception: Exception?) {
        speechService = null
        val errorMessage = "Erreur Vosk: ${exception?.message}"
        Log.e(TAG, "Erreur dans le listener STT", exception)
        _result.postValue(SttResult(SttStatus.ERROR, errorMessage))
    }

    override fun onTimeout() {
        speechService = null
        _result.postValue(SttResult(SttStatus.IDLE, "Timeout. $currentFinalText", isFinal = true))
        Log.w(TAG, "Vosk onTimeout: Arrêt de l'écoute.")
    }
}

package be.aktina.android.kikkopocnearby.services

import android.content.Context
import android.speech.tts.TextToSpeech
import android.speech.tts.UtteranceProgressListener
import android.util.Log
import java.util.Locale
import java.util.UUID

/**
 * FRELON-ARCHITECTE: Service singleton pour gérer le Text-To-Speech (TTS).
 * Cette classe encapsule l'initialisation et l'utilisation du moteur TTS d'Android,
 * le rendant réutilisable et facile à gérer depuis n'importe quelle Activity.
 * C'est notre "Abeille-Annonceuse".
 */
object TtsService : TextToSpeech.OnInitListener {

    private const val TAG = "TtsService"
    private var tts: TextToSpeech? = null
    private var isInitialized = false
    private var isAvailable = false
    private val pendingQueue = mutableListOf<Pair<String, String>>()

    /**
     * Initialise le service TTS. Doit être appelé une seule fois, idéalement depuis
     * une classe Application ou la première Activity.
     */
    fun initialize(context: Context) {
        if (isInitialized) {
            Log.v(TAG, "TTS est déjà en cours d'initialisation ou initialisé.")
            return
        }
        Log.v(TAG, "Initialisation du service TTS...")
        tts = TextToSpeech(context.applicationContext, this)
        isInitialized = true
    }

    override fun onInit(status: Int) {
        if (status == TextToSpeech.SUCCESS) {
            isAvailable = true
            tts?.language = Locale.getDefault()
            Log.i(TAG, "TTS initialisé avec succès pour la langue: ${Locale.getDefault().displayName}")
            // Si des messages étaient en attente, on les lit maintenant.
            processPendingQueue()
        } else {
            isAvailable = false
            Log.e(TAG, "Échec de l'initialisation du TTS. Statut: $status")
        }
    }

    /**
     * Fait parler le Bourdon. Si le TTS n'est pas prêt, le message est mis en file d'attente.
     */
    fun speak(text: String, utteranceId: String = UUID.randomUUID().toString()) {
        if (isAvailable) {
            Log.d(TAG, "Parole demandée: '$text' (ID: $utteranceId)")
            tts?.speak(text, TextToSpeech.QUEUE_FLUSH, null, utteranceId)
        } else {
            Log.w(TAG, "TTS non disponible, mise en file d'attente du message (ID: $utteranceId)")
            pendingQueue.add(Pair(text, utteranceId))
        }
    }

    /**
     * Définit un listener pour suivre la progression de la parole.
     */
    fun setProgressListener(listener: UtteranceProgressListener) {
        if (isAvailable) {
            tts?.setOnUtteranceProgressListener(listener)
        } else {
            Log.w(TAG, "Impossible de définir le ProgressListener : TTS non initialisé.")
        }
    }

    private fun processPendingQueue() {
        if (pendingQueue.isNotEmpty()) {
            Log.i(TAG, "Traitement de ${pendingQueue.size} message(s) en attente...")
            val queueCopy = pendingQueue.toList()
            pendingQueue.clear()
            queueCopy.forEach { (text, id) ->
                speak(text, id)
            }
        }
    }

    /**
     * Libère les ressources du service TTS. À appeler dans le onDestroy de l'application.
     */
    fun shutdown() {
        Log.v(TAG, "Arrêt du service TTS.")
        isInitialized = false
        isAvailable = false
        tts?.stop()
        tts?.shutdown()
        tts = null
    }
}

package be.aktina.android.kikkopocnearby.ui

import android.annotation.SuppressLint

class ModelAdapter(
    private var models: List<Model>,
    private val onActionClickListener: (Model) -> Unit
) : RecyclerView.Adapter<ModelAdapter.ModelViewHolder>() {

    // MODIFIÉ : Le Payload devient une classe pour transporter notre état de décompression.
    class ProgressPayload(val isUnzipping: Boolean = false)

    class ModelViewHolder(view: View) : RecyclerView.ViewHolder(view) {
        val modelName: TextView = view.findViewById(R.id.model_name)
        val modelDescription: TextView = view.findViewById(R.id.model_description)
        val actionButton: ImageButton = view.findViewById(R.id.model_action_button)
        val progressContainer: LinearLayout = view.findViewById(R.id.download_progress_container)
        val progressBar: ProgressBar = view.findViewById(R.id.download_progress_bar)
        val progressText: TextView = view.findViewById(R.id.download_progress_text)
    }

    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): ModelViewHolder {
        val view = LayoutInflater.from(parent.context)
            .inflate(R.layout.item_model, parent, false)
        return ModelViewHolder(view)
    }

    override fun getItemCount() = models.size

    // MODIFIÉ : Logique de mise à jour partielle améliorée.
    override fun onBindViewHolder(holder: ModelViewHolder, position: Int, payloads: MutableList<Any>) {
        if (payloads.isEmpty()) {
            // Si pas de payload, c'est un affichage complet.
            super.onBindViewHolder(holder, position, payloads)
            return
        }

        // On cherche notre payload spécifique.
        val progressPayload = payloads.find { it is ProgressPayload } as? ProgressPayload
        if (progressPayload != null) {
            // Si on le trouve, on met à jour uniquement la progression.
            bindProgress(holder, models[position], progressPayload.isUnzipping)
        } else {
            // Pour tout autre payload (non utilisé ici), on ferait un refresh complet.
            super.onBindViewHolder(holder, position, payloads)
        }
    }

    override fun onBindViewHolder(holder: ModelViewHolder, position: Int) {
        val model = models[position]
        bind(holder, model)
    }

    private fun bind(holder: ModelViewHolder, model: Model) {
        holder.modelName.text = model.name
        holder.modelDescription.text = model.description
        holder.actionButton.setOnClickListener { onActionClickListener(model) }
        bindProgress(holder, model)
    }

    // MODIFIÉ : La fonction accepte maintenant l'état 'isUnzipping'.
    @SuppressLint("SetTextI18n")
    private fun bindProgress(holder: ModelViewHolder, model: Model, isUnzipping: Boolean = false) {
        when (model.status) {
            DownloadStatus.NOT_DOWNLOADED -> {
                holder.progressContainer.visibility = View.GONE
                holder.actionButton.visibility = View.VISIBLE
                holder.actionButton.setImageResource(R.drawable.ic_download)
                holder.actionButton.isEnabled = true
            }
            DownloadStatus.DOWNLOADING -> {
                holder.progressContainer.visibility = View.VISIBLE
                holder.actionButton.visibility = View.GONE

                // NOUVELLE LOGIQUE D'AFFICHAGE
                if (isUnzipping) {
                    holder.progressBar.isIndeterminate = true
                    holder.progressText.text = "Décompression en cours..."
                } else {
                    holder.progressBar.isIndeterminate = model.sizeInBytes <= 0 // Indéterminé si taille inconnue
                    holder.progressBar.progress = model.progress

                    if (model.sizeInBytes > 0) {
                        val downloadedMb = model.downloadedBytes / (1024.0 * 1024.0)
                        val totalMb = model.sizeInBytes / (1024.0 * 1024.0)
                        holder.progressText.text = String.format(Locale.getDefault(), "%.1f MB / %.1f MB (%d%%)", downloadedMb, totalMb, model.progress)
                    } else {
                        // Cas où la taille totale n'est pas connue
                        val downloadedMb = model.downloadedBytes / (1024.0 * 1024.0)
                        holder.progressText.text = String.format(Locale.getDefault(), "%.1f MB téléchargés", downloadedMb)
                    }
                }
            }
            DownloadStatus.SUCCEEDED -> {
                holder.progressContainer.visibility = View.GONE
                holder.actionButton.visibility = View.VISIBLE
                holder.actionButton.setImageResource(R.drawable.ic_delete)
                holder.actionButton.isEnabled = true
            }
            DownloadStatus.FAILED -> {
                holder.progressContainer.visibility = View.VISIBLE
                holder.progressBar.visibility = View.GONE
                holder.actionButton.visibility = View.VISIBLE
                holder.actionButton.setImageResource(R.drawable.ic_download)
                holder.actionButton.isEnabled = true
                holder.progressText.text = "Échec du téléchargement"
            }
        }
    }

    @SuppressLint("NotifyDataSetChanged")
    fun updateModels(newModels: List<Model>) {
        this.models = newModels
        notifyDataSetChanged()
    }
}

package be.aktina.android.kikkopocnearby.workers

import android.content.Context
import android.util.Log
import androidx.work.CoroutineWorker
import androidx.work.WorkerParameters
import androidx.work.workDataOf
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import org.apache.commons.compress.archivers.tar.TarArchiveInputStream
import org.apache.commons.compress.compressors.gzip.GzipCompressorInputStream
import java.io.BufferedInputStream
import java.io.File
import java.io.FileInputStream
import java.io.FileOutputStream
import java.io.IOException
import java.net.HttpURLConnection
import java.net.URL
import java.util.zip.ZipInputStream

class DownloadWorker(
    appContext: Context,
    params: WorkerParameters
) : CoroutineWorker(appContext, params) {

    companion object {
        const val KEY_MODEL_ID = "model_id"
        const val KEY_DOWNLOAD_URL = "download_url"
        const val KEY_ARCHIVE_PATH = "archive_path"
        const val KEY_IS_ARCHIVE = "is_archive"

        const val KEY_PROGRESS_BYTES = "progress_bytes"
        const val KEY_PROGRESS_PERCENT = "progress_percent"

        // NOUVELLE CLÉ POUR SUIVRE L'ÉTAT DE DÉCOMPRESSION
        const val KEY_START_UNZIPPING = "start_unzipping"
    }

    private val TAG = "DownloadWorker"
    private val PROGRESS_UPDATE_THRESHOLD = 5

    override suspend fun doWork(): Result {
        val modelId = inputData.getString(KEY_MODEL_ID) ?: return Result.failure()
        val downloadUrl = inputData.getString(KEY_DOWNLOAD_URL) ?: return Result.failure()
        val archivePath = inputData.getString(KEY_ARCHIVE_PATH) ?: return Result.failure()
        val isArchive = inputData.getBoolean(KEY_IS_ARCHIVE, false)

        Log.i(TAG, "Démarrage du téléchargement pour '$modelId' depuis $downloadUrl vers $archivePath")
        val outputFile = File(archivePath)

        return withContext(Dispatchers.IO) {
            try {
                downloadFile(downloadUrl, outputFile)

                if (isArchive) {
                    val destinationDir = outputFile.parentFile
                    if (destinationDir == null) {
                        Log.e(TAG, "Dossier parent invalide pour la décompression.")
                        return@withContext Result.failure()
                    }

                    // ON PRÉVIENT L'UI QU'ON COMMENCE LA DÉCOMPRESSION
                    setProgress(workDataOf(KEY_START_UNZIPPING to true))

                    Log.i(TAG, "Le fichier est une archive. Démarrage de la décompression...")

                    if (outputFile.extension.equals("zip", ignoreCase = true)) {
                        unzip(outputFile, destinationDir)
                        Log.i(TAG, "Décompression ZIP terminée dans ${destinationDir.absolutePath}")
                    } else if (outputFile.name.endsWith(".tar.gz", ignoreCase = true)) {
                        untar(outputFile, destinationDir)
                        Log.i(TAG, "Décompression TAR.GZ terminée dans ${destinationDir.absolutePath}")
                    }

                    outputFile.delete()
                    Log.d(TAG, "Archive ${outputFile.name} supprimée.")
                }

                Log.i(TAG, "Tâche terminée avec succès pour le modèle '$modelId'.")
                Result.success()
            } catch (e: Exception) {
                Log.e(TAG, "Échec de la tâche pour le modèle '$modelId'", e)
                if (outputFile.exists()) outputFile.deleteRecursively()
                Result.failure()
            }
        }
    }

    @Throws(IOException::class)
    private suspend fun downloadFile(downloadUrl: String, outputFile: File) {
        var connection: HttpURLConnection? = null
        var inputStream: java.io.InputStream? = null
        var outputStream: FileOutputStream? = null

        try {
            val url = URL(downloadUrl)
            connection = url.openConnection() as HttpURLConnection
            connection.connect()

            if (connection.responseCode !in 200..299) {
                throw IOException("Erreur serveur: ${connection.responseCode} ${connection.responseMessage}")
            }

            val totalBytes = connection.contentLength.toLong()
            inputStream = connection.inputStream
            outputStream = FileOutputStream(outputFile)
            val buffer = ByteArray(8 * 1024)
            var bytesCopied: Long = 0
            var bytesRead: Int
            var lastReportedProgress = -1

            while (inputStream.read(buffer).also { bytesRead = it } != -1) {
                if (isStopped) throw InterruptedException("Tâche de téléchargement annulée.")
                outputStream.write(buffer, 0, bytesRead)
                bytesCopied += bytesRead
                if (totalBytes > 0) {
                    val progressPercent = ((bytesCopied * 100) / totalBytes).toInt()
                    if (progressPercent >= lastReportedProgress + PROGRESS_UPDATE_THRESHOLD) {
                        lastReportedProgress = progressPercent
                        setProgress(workDataOf(KEY_PROGRESS_BYTES to bytesCopied, KEY_PROGRESS_PERCENT to progressPercent))
                    }
                }
            }
            // S'assure que la progression atteigne bien 100% si la taille est connue
            if (totalBytes > 0) {
                setProgress(workDataOf(KEY_PROGRESS_BYTES to totalBytes, KEY_PROGRESS_PERCENT to 100))
            }
        } finally {
            outputStream?.close()
            inputStream?.close()
            connection?.disconnect()
        }
    }

    @Throws(IOException::class)
    private fun unzip(zipFile: File, targetDirectory: File) {
        val tempUnzipDir = File(zipFile.parent, "temp_unzip_${System.currentTimeMillis()}")
        tempUnzipDir.mkdirs()

        ZipInputStream(BufferedInputStream(FileInputStream(zipFile))).use { zis ->
            var entry = zis.nextEntry
            while (entry != null) {
                val newFile = File(tempUnzipDir, entry.name)
                if (!newFile.canonicalPath.startsWith(tempUnzipDir.canonicalPath + File.separator)) {
                    throw IOException("Zip Slip vulnerability detected!")
                }

                if (entry.isDirectory) {
                    if (!newFile.isDirectory && !newFile.mkdirs()) {
                        throw IOException("Failed to create directory ${newFile.absolutePath}")
                    }
                } else {
                    newFile.parentFile?.mkdirs()
                    FileOutputStream(newFile).use { fos -> zis.copyTo(fos) }
                }
                zis.closeEntry()
                entry = zis.nextEntry
            }
        }

        val filesInTemp = tempUnzipDir.listFiles()
        if (filesInTemp != null && filesInTemp.size == 1 && filesInTemp[0].isDirectory) {
            val rootDirInZip = filesInTemp[0]
            rootDirInZip.listFiles()?.forEach { file ->
                file.renameTo(File(targetDirectory, file.name))
            }
        } else {
            filesInTemp?.forEach { file ->
                file.renameTo(File(targetDirectory, file.name))
            }
        }

        tempUnzipDir.deleteRecursively()
    }

    @Throws(IOException::class)
    private fun untar(tarGzFile: File, targetDirectory: File) {
        val fis = FileInputStream(tarGzFile)
        val bis = BufferedInputStream(fis)
        val gzis = GzipCompressorInputStream(bis)
        val taris = TarArchiveInputStream(gzis)

        taris.use {
            var entry = it.nextEntry
            while (entry != null) {
                val newFile = File(targetDirectory, entry.name)
                if (!newFile.canonicalPath.startsWith(targetDirectory.canonicalPath + File.separator)) {
                    throw IOException("Tar Slip vulnerability detected!")
                }

                if (entry.isDirectory) {
                    newFile.mkdirs()
                } else {
                    newFile.parentFile?.mkdirs()
                    FileOutputStream(newFile).use { fos -> it.copyTo(fos) }
                }
                entry = it.nextEntry
            }
        }
    }
}
